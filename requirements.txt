# evogym must be installed separately from source (see evogym/ in this repo).
# setuptools<81: evogym/envs/base.py still does `import pkg_resources`, which
# setuptools dropped as of 84.0.0 (deprecated, then removed). Without this
# pin, a fresh venv rebuild fails at `import evogym.envs` with
# ModuleNotFoundError: No module named 'pkg_resources'. See
# docs/LAB_MACHINE_SETUP.md Part 4 / risk register item 2.
setuptools<81
numpy<2.0.0
gymnasium
stable-baselines3
pyyaml
pytest
pandas
matplotlib
# imageio-ffmpeg bundles a portable ffmpeg binary, so mp4 writing doesn't
# depend on a system ffmpeg install. pygifsicle (gif optimization) is
# deliberately NOT a dependency here -- it shells out to an external
# gifsicle binary that isn't guaranteed to exist on every machine; gif
# optimization is optional and degrades gracefully without it (see
# evodevo/visualization/render.py).
imageio
imageio-ffmpeg
